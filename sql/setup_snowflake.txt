-- Create role
CREATE ROLE IF NOT EXISTS GLOBO_ROLE;

-- Grant role to your user
GRANT ROLE GLOBO_ROLE TO USER VICTORIAGEORGIEVA8;

-- Create warehouse
CREATE WAREHOUSE IF NOT EXISTS COMPUTE_WH
  WAREHOUSE_SIZE = 'XSMALL'
  AUTO_SUSPEND = 60
  AUTO_RESUME = TRUE;

-- Create database
CREATE DATABASE IF NOT EXISTS GLOBO_DB;

-- Create schemas
CREATE SCHEMA IF NOT EXISTS RAW;
CREATE SCHEMA IF NOT EXISTS CLEANSED;
CREATE SCHEMA IF NOT EXISTS STAR;
CREATE SCHEMA IF NOT EXISTS PRESENTATION;

-- Create file format
CREATE OR REPLACE FILE FORMAT RAW.CSV_FORMAT
  TYPE = 'CSV'
  FIELD_DELIMITER = ','
  SKIP_HEADER = 1;

-- Create stage for S3
CREATE OR REPLACE STAGE RAW.S3_STAGE
  URL = 's3://data-warehouse-etll/processed/'
  FILE_FORMAT = RAW.CSV_FORMAT;

-- Create table for cleansed sales
CREATE OR REPLACE TABLE CLEANSED.SALES_CLEAN (
    SALES_ID INT,
    PRODUCT_ID INT,
    REGION STRING,
    QTY INT,
    PRICE FLOAT,
    TIME_STAMP TIMESTAMP,
    DISCOUNT FLOAT,
    ORDER_STATUS STRING,
    CATEGORY STRING,
    BRAND STRING,
    RATING FLOAT,
    IN_STOCK BOOLEAN,
    LAUNCH_DATE DATE
);
-- DIM_DATE
CREATE OR REPLACE TABLE STAR.DIM_DATE AS
SELECT
    DISTINCT TIME_STAMP::DATE AS DATE,
    YEAR(TIME_STAMP) AS YEAR,
    MONTH(TIME_STAMP) AS MONTH,
    DAY(TIME_STAMP) AS DAY,
    TO_CHAR(TIME_STAMP, 'MON') AS MONTH_NAME
FROM CLEANSED.SALES_CLEAN;

-- DIM_PRODUCT
CREATE OR REPLACE TABLE STAR.DIM_PRODUCT AS
SELECT
    DISTINCT PRODUCT_ID,
    CATEGORY,
    BRAND,
    RATING,
    IN_STOCK,
    LAUNCH_DATE
FROM CLEANSED.SALES_CLEAN;

-- FACT_SALES
CREATE OR REPLACE TABLE STAR.FACT_SALES AS
SELECT
    SALES_ID,
    PRODUCT_ID,
    REGION,
    QTY,
    PRICE,
    DISCOUNT,
    ORDER_STATUS,
    TIME_STAMP::DATE AS DATE
FROM CLEANSED.SALES_CLEAN;
-- Average order value by region
CREATE OR REPLACE MATERIALIZED VIEW PRESENTATION.MV_AVG_ORDER_VALUE_BY_REGION AS
SELECT
    REGION,
    AVG(QTY * PRICE) AS AVG_ORDER_VALUE
FROM STAR.FACT_SALES
GROUP BY REGION;

-- Top brands by quantity
CREATE OR REPLACE MATERIALIZED VIEW PRESENTATION.MV_TOP_BRANDS_BY_QUANTITY AS
SELECT
    BRAND,
    SUM(QTY) AS TOTAL_QTY
FROM CLEANSED.SALES_CLEAN
GROUP BY BRAND
ORDER BY TOTAL_QTY DESC;

-- Monthly revenue growth
CREATE OR REPLACE MATERIALIZED VIEW PRESENTATION.MV_MONTHLY_REVENUE_GROWTH AS
SELECT
    YEAR(TIME_STAMP) AS YEAR,
    MONTH(TIME_STAMP) AS MONTH,
    SUM(QTY * PRICE) AS REVENUE
FROM CLEANSED.SALES_CLEAN
GROUP BY YEAR, MONTH
ORDER BY YEAR, MONTH;

-- Category margin analysis
CREATE OR REPLACE MATERIALIZED VIEW PRESENTATION.MV_CATEGORY_MARGIN_ANALYSIS AS
SELECT
    CATEGORY,
    SUM(QTY * PRICE) AS TOTAL_REVENUE,
    SUM(QTY * (PRICE - (PRICE * DISCOUNT))) AS MARGIN
FROM CLEANSED.SALES_CLEAN
GROUP BY CATEGORY;
